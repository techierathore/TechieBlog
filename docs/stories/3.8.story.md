```csharp
public IEnumerable<BlogPost> GetPublishedPosts(int pageSize, int offset)
{
    using var vConn = GetOpenConnection();
    var vParams = new DynamicParameters();
    vParams.Add("@PageSize", pageSize);
    vParams.Add("@Offset", offset);
    return vConn.Query<BlogPost>(
        "SELECT * FROM GetPublishedPosts(@PageSize, @Offset)",
        vParams
    ).ToList();
}

public BlogPost GetBySlug(string slug)
{
    using var vConn = GetOpenConnection();
    var vParams = new DynamicParameters();
    vParams.Add("@Slug", slug);
    return vConn.Query<BlogPost>(
        "SELECT * FROM GetPostBySlug(@Slug)",
        vParams
    ).FirstOrDefault();
}
```

### PostgreSQL Functions to Create/Verify

Check BlogDb/PostgresScripts for these, create if missing:

```sql
-- Get published posts with pagination
CREATE OR REPLACE FUNCTION GetPublishedPosts(pageSize INT, offset INT)
RETURNS SETOF BlogPost AS $$
BEGIN
    RETURN QUERY
    SELECT * FROM BlogPost
    WHERE Published = true AND (IsDeleted IS NULL OR IsDeleted = false)
    ORDER BY CreatedOn DESC
    LIMIT pageSize OFFSET offset;
END;
$$ LANGUAGE plpgsql;

-- Get post by slug
CREATE OR REPLACE FUNCTION GetPostBySlug(slug VARCHAR)
RETURNS SETOF BlogPost AS $$
BEGIN
    RETURN QUERY
    SELECT * FROM BlogPost
    WHERE Slug = slug AND Published = true;
END;
$$ LANGUAGE plpgsql;
```

### Reading Time Calculator Pattern

```csharp
public static class ReadingTimeCalculator
{
    private const int WordsPerMinute = 200;

    public static string Calculate(string content)
    {
        if (string.IsNullOrWhiteSpace(content))
            return "1 min read";

        var wordCount = content.Split(
            new[] { ' ', '\n', '\r', '\t' },
            StringSplitOptions.RemoveEmptyEntries
        ).Length;

        var minutes = Math.Max(1, (int)Math.Ceiling(wordCount / (double)WordsPerMinute));
        return $"{minutes} min read";
    }
}
```

### Home.razor Data Loading Pattern

```csharp
@inject BlogSvc BlogService

@code {
    private List<BlogPost> recentPosts = new();
    private BlogPost featuredPost = new();
    private int currentPage = 1;
    private int totalPages = 1;
    private const int PageSize = 6;
    private bool isLoading = true;

    protected override async Task OnInitializedAsync()
    {
        await LoadPosts();
    }

    private async Task LoadPosts()
    {
        isLoading = true;
        var offset = (currentPage - 1) * PageSize;

        // Get featured (first post)
        featuredPost = BlogService.GetFeaturedPost();

        // Get recent (skip featured)
        recentPosts = BlogService.GetPublishedPosts(PageSize, offset)
            .Where(p => p.PostID != featuredPost?.PostID)
            .ToList();

        isLoading = false;
        StateHasChanged();
    }

    private async Task HandlePageChanged(int page)
    {
        currentPage = page;
        await LoadPosts();
    }
}
```

### PostView.razor Data Loading Pattern

```csharp
@inject BlogSvc BlogService
@inject NavigationManager Navigation

@code {
    [Parameter]
    public string Slug { get; set; } = string.Empty;

    private BlogPost post = new();
    private bool isLoading = true;
    private bool notFound = false;

    protected override async Task OnParametersSetAsync()
    {
        await LoadPost();
    }

    private async Task LoadPost()
    {
        isLoading = true;
        post = BlogService.GetPostBySlug(Slug);

        if (post == null)
        {
            notFound = true;
        }

        isLoading = false;
        StateHasChanged();
    }
}
```
