### Database Schema

```sql
-- PostgreSQL migration script
CREATE TABLE IF NOT EXISTS UserFavorite (
    FavoriteId BIGSERIAL PRIMARY KEY,
    PostId BIGINT NOT NULL REFERENCES BlogPost(PostID) ON DELETE CASCADE,
    UserId BIGINT NOT NULL REFERENCES BlogUser(UserId) ON DELETE CASCADE,
    CreatedOn TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT UQ_UserFavorite_User_Post UNIQUE (PostId, UserId)
);

CREATE INDEX IX_UserFavorite_PostId ON UserFavorite(PostId);
CREATE INDEX IX_UserFavorite_UserId ON UserFavorite(UserId);
CREATE INDEX IX_UserFavorite_CreatedOn ON UserFavorite(UserId, CreatedOn DESC);
```

### Service Pattern

```csharp
public class FavoriteSvc
{
    private readonly IFavoriteRepo _favoriteRepo;
    private readonly IBlogPostRepo _postRepo;
    
    public Result AddFavorite(long postId, long userId)
    {
        var existing = _favoriteRepo.GetByPostAndUser(postId, userId);
        if (existing != null)
            return Result.Success(); // Already favorited
            
        _favoriteRepo.Insert(new UserFavorite
        {
            PostId = postId,
            UserId = userId,
            CreatedOn = DateTime.UtcNow
        });
        return Result.Success();
    }
    
    public Result RemoveFavorite(long postId, long userId)
    {
        _favoriteRepo.Delete(postId, userId);
        return Result.Success();
    }
    
    public bool IsFavorited(long postId, long userId)
    {
        return _favoriteRepo.GetByPostAndUser(postId, userId) != null;
    }
    
    public IEnumerable<BlogPost> GetUserFavorites(long userId)
    {
        var favorites = _favoriteRepo.GetByUser(userId);
        // Join with posts or return post IDs for separate lookup
        return favorites.Select(f => _postRepo.GetSingle(f.PostId)).Where(p => p != null);
    }
}
```

### UI Component Pattern

```razor
@* FavoriteToggle.razor *@
<button class="favorite-toggle @(IsFavorited ? "favorited" : "")" 
        @onclick="Toggle" 
        disabled="@(!IsLoggedIn)">
    @if (IsFavorited)
    {
        <FluentIcon Value="@(new Icons.Filled.Size20.Heart())" Color="Color.Error" />
    }
    else
    {
        <FluentIcon Value="@(new Icons.Regular.Size20.Heart())" />
    }
</button>

@code {
    [Parameter] public long PostId { get; set; }
    [Parameter] public bool IsFavorited { get; set; }
    [Parameter] public bool IsLoggedIn { get; set; }
    [Parameter] public EventCallback<bool> OnToggle { get; set; }
    
    private async Task Toggle()
    {
        IsFavorited = !IsFavorited;
        await OnToggle.InvokeAsync(IsFavorited);
    }
}
```
