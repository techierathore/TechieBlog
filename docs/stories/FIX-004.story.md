### Repository Implementation

Add to `BlogPostRepo.cs`:
```csharp
/// <summary>
/// Searches posts by keyword in Title, Abstract, and PostContent.
/// </summary>
public IEnumerable<BlogPost> SearchPosts(string query, int limit = 10, int offset = 0)
{
    if (string.IsNullOrWhiteSpace(query))
        return Enumerable.Empty<BlogPost>();
    
    using var vConn = GetOpenConnection();
    const string sql = @"
        SELECT p.PostID, p.Title, p.Slug, p.Abstract, p.PostContent, p.CreatedOn, p.UpdatedOn,
               p.UserID, p.Tags, p.CategoryId, p.FeaturedImage, p.Published,
               CONCAT(u.FirstName, ' ', u.LastName) as BlogWriter
        FROM BlogPost p
        LEFT JOIN BlogUser u ON p.UserID = u.UserId
        WHERE p.Published = TRUE 
            AND (p.IsDeleted = FALSE OR p.IsDeleted IS NULL)
            AND (p.Title ILIKE @SearchPattern 
                 OR p.Abstract ILIKE @SearchPattern 
                 OR p.PostContent ILIKE @SearchPattern)
        ORDER BY p.CreatedOn DESC
        LIMIT @Limit OFFSET @Offset";
    
    return vConn.Query<BlogPost>(sql, new { 
        SearchPattern = $"%{query}%", 
        Limit = limit, 
        Offset = offset 
    }).ToList();
}

/// <summary>
/// Gets count of posts matching search query.
/// </summary>
public int GetSearchResultCount(string query)
{
    if (string.IsNullOrWhiteSpace(query))
        return 0;
    
    using var vConn = GetOpenConnection();
    const string sql = @"
        SELECT COUNT(*) FROM BlogPost
        WHERE Published = TRUE 
            AND (IsDeleted = FALSE OR IsDeleted IS NULL)
            AND (Title ILIKE @SearchPattern 
                 OR Abstract ILIKE @SearchPattern 
                 OR PostContent ILIKE @SearchPattern)";
    
    return vConn.ExecuteScalar<int>(sql, new { SearchPattern = $"%{query}%" });
}
```
