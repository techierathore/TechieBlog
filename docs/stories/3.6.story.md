## Story

**As an** author,
**I want** to schedule posts for future publication,
**so that** I can prepare content in advance and publish at optimal times.

4. Scheduled posts visible in admin with "Scheduled" status
5. Author can edit/cancel scheduled posts
6. Scheduled posts not visible on public pages until published
7. Timezone handling for consistent scheduling

### BlogPost Model Extension

```csharp
public class BlogPost
{
    // Existing fields...

    /// <summary>
    /// UTC timestamp when post is scheduled to publish (null if not scheduled)
    /// </summary>
    public DateTime? ScheduledPublishOn { get; set; }

    /// <summary>
    /// UTC timestamp when post was actually published
    /// </summary>
    public DateTime? PublishedOn { get; set; }

    /// <summary>
    /// Returns true if post is scheduled for future publication
    /// </summary>
    public bool IsScheduled => !Published && ScheduledPublishOn.HasValue && ScheduledPublishOn > DateTime.UtcNow;

    /// <summary>
    /// Returns the current status of the post
    /// </summary>
    public string Status => Published ? "Published" : IsScheduled ? "Scheduled" : "Draft";
}
```

```razor
    </FluentCard>

@code {
    private DateTime? scheduledDate;
    private DateTime? scheduledTime;

    private DateTime? GetScheduledDateTime()
    {
        if (!scheduledDate.HasValue) return null;

        var date = scheduledDate.Value;
        if (scheduledTime.HasValue)
        {
            date = date.Date.Add(scheduledTime.Value.TimeOfDay);
        }
        return date;
    }

    private async Task SchedulePost()
    {
        var scheduledDateTime = GetScheduledDateTime();
        if (!scheduledDateTime.HasValue)
        {
            errorMessage = "Please select a date and time for scheduling";
            return;
        }

        if (scheduledDateTime <= DateTime.Now)
        {
            errorMessage = "Scheduled time must be in the future";
            return;
        }

        PageObj.ScheduledPublishOn = scheduledDateTime.Value.ToUniversalTime();
        PageObj.Published = false;

        await SavePost();
        successMessage = $"Post scheduled for {scheduledDateTime:f}";
    }
}
```

### Background Publishing Service Pattern

```csharp
using Microsoft.Extensions.Hosting;
using Microsoft.Extensions.Logging;
using Microsoft.Extensions.DependencyInjection;

namespace BlogEngine.Services;

public class ScheduledPostPublisher : BackgroundService
{
    private readonly IServiceProvider _services;
    private readonly ILogger<ScheduledPostPublisher> _logger;
    private readonly TimeSpan _checkInterval = TimeSpan.FromMinutes(1);

    public ScheduledPostPublisher(
        IServiceProvider services,
        ILogger<ScheduledPostPublisher> logger)
    {
        _services = services;
        _logger = logger;
    }

    protected override async Task ExecuteAsync(CancellationToken stoppingToken)
    {
        _logger.LogInformation("Scheduled Post Publisher started");

        while (!stoppingToken.IsCancellationRequested)
        {
            await PublishDuePosts();
            await Task.Delay(_checkInterval, stoppingToken);
        }
    }

    private async Task PublishDuePosts()
    {
        try
        {
            using var scope = _services.CreateScope();
            var blogSvc = scope.ServiceProvider.GetRequiredService<BlogSvc>();

            var duePosts = blogSvc.GetDueScheduledPosts();
            foreach (var post in duePosts)
            {
                post.Published = true;
                post.PublishedOn = post.ScheduledPublishOn ?? DateTime.UtcNow;
                post.ScheduledPublishOn = null;

                blogSvc.UpdatePost(post);
                _logger.LogInformation($"Published scheduled post: {post.Title} (ID: {post.PostID})");
            }
        }
        catch (Exception ex)
        {
            _logger.LogError(ex, "Error publishing scheduled posts");
        }
    }
}
```

### BlogSvc Extension Pattern

```csharp
public class BlogSvc
{
    public List<BlogPost> GetDueScheduledPosts()
    {
        // Get posts where ScheduledPublishOn <= now and not published
        return PostRepo.GetDueScheduledPosts(DateTime.UtcNow);
    }

    public List<BlogPost> GetScheduledPosts()
    {
        // Get all scheduled posts for admin view
        return PostRepo.GetScheduledPosts();
    }

    public void SchedulePost(BlogPost post, DateTime scheduledUtc)
    {
        post.ScheduledPublishOn = scheduledUtc;
        post.Published = false;
        post.UpdatedOn = DateTime.UtcNow;

        if (post.PostID == 0)
            PostRepo.Insert(post);
        else
            PostRepo.Update(post);
    }

    public void CancelSchedule(long postId)
    {
        var post = PostRepo.GetSingle(postId);
        if (post != null)
        {
            post.ScheduledPublishOn = null;
            post.UpdatedOn = DateTime.UtcNow;
            PostRepo.Update(post);
        }
    }
}
```
