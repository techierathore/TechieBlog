### Key Integration Points

1. **AuthSvc.AppSignUp()** expects encrypted SvcData - may need simpler RegisterUser() method
2. **Password hashing** via AppEncrypt.CreateHash() - SHA256 based
3. **After registration** - user should login (don't auto-login for security)

### Recommended Approach

Create a simpler RegisterUser() method that doesn't require encryption:

```csharp
public Result<AppUser> RegisterUser(string displayName, string email, string password)
{
    // Validate email uniqueness
    var existing = UserRepo.GetUserByEmail(email);
    if (existing != null)
        return Result<AppUser>.Failure("Email already registered");

    // Hash password
    var hashedPassword = AppEncrypt.CreateHash(password);

    // Create user
    var newUser = new AppUser
    {
        FullName = displayName,
        EmailId = email,
        LoginPass = hashedPassword,
        UserRole = AppConstants.AppUseRole,
        CreatedOn = DateTime.UtcNow,
        IsConfirmed = false // Email not verified yet
    };

    var userId = UserRepo.InsertToGetId(newUser);
    if (userId <= 0)
        return Result<AppUser>.Failure("Registration failed. Please try again.");

    newUser.UserId = userId;
    return Result<AppUser>.Success(newUser);
}
```

### Password Validator Pattern

```csharp
namespace BlogEngine.Common;

public static class PasswordValidator
{
    public static ValidationResult Validate(string password)
    {
        var errors = new List<string>();

        if (string.IsNullOrEmpty(password))
        {
            errors.Add("Password is required");
            return new ValidationResult(false, errors);
        }

        if (password.Length < 8)
            errors.Add("Password must be at least 8 characters");

        if (!password.Any(char.IsUpper))
            errors.Add("Password must contain an uppercase letter");

        if (!password.Any(char.IsLower))
            errors.Add("Password must contain a lowercase letter");

        if (!password.Any(char.IsDigit))
            errors.Add("Password must contain a number");

        return new ValidationResult(errors.Count == 0, errors);
    }
}

public record ValidationResult(bool IsValid, List<string> Errors);
```

### RegisterPage Implementation Pattern

```csharp
@inject AuthSvc AuthService
@inject NavigationManager Navigation

private bool isProcessing = false;
private string successMessage = string.Empty;

private async Task HandleSubmit()
{
    errorMessage = string.Empty;
    successMessage = string.Empty;

    // Validate terms
    if (!model.AgreeToTerms)
    {
        errorMessage = "You must agree to the Terms of Service";
        return;
    }

    // Validate password match
    if (model.Password != model.ConfirmPassword)
    {
        errorMessage = "Passwords do not match";
        return;
    }

    // Validate password strength
    var passwordResult = PasswordValidator.Validate(model.Password);
    if (!passwordResult.IsValid)
    {
        errorMessage = string.Join(". ", passwordResult.Errors);
        return;
    }

    isProcessing = true;
    StateHasChanged();

    try
    {
        var result = AuthService.RegisterUser(
            model.DisplayName,
            model.Email,
            model.Password
        );

        if (result.IsSuccess)
        {
            successMessage = "Account created successfully! Redirecting to login...";
            StateHasChanged();
            await Task.Delay(2000);
            Navigation.NavigateTo("/login");
        }
        else
        {
            errorMessage = result.ErrorMessage;
        }
    }
    catch (Exception ex)
    {
        errorMessage = "An error occurred. Please try again.";
    }
    finally
    {
        isProcessing = false;
        StateHasChanged();
    }
}
```

### DataAnnotations for RegisterModel

```csharp
using System.ComponentModel.DataAnnotations;

private class RegisterModel
{
    [Required(ErrorMessage = "Display name is required")]
    [StringLength(100, ErrorMessage = "Display name cannot exceed 100 characters")]
    public string DisplayName { get; set; } = string.Empty;

    [Required(ErrorMessage = "Email is required")]
    [EmailAddress(ErrorMessage = "Invalid email format")]
    public string Email { get; set; } = string.Empty;

    [Required(ErrorMessage = "Password is required")]
    [MinLength(8, ErrorMessage = "Password must be at least 8 characters")]
    public string Password { get; set; } = string.Empty;

    [Required(ErrorMessage = "Please confirm your password")]
    [Compare(nameof(Password), ErrorMessage = "Passwords do not match")]
    public string ConfirmPassword { get; set; } = string.Empty;

    public bool AgreeToTerms { get; set; }
}
```
