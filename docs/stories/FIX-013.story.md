### Database Schema

```sql
-- PostgreSQL migration script
CREATE TABLE IF NOT EXISTS PostRating (
    RatingId BIGSERIAL PRIMARY KEY,
    PostId BIGINT NOT NULL REFERENCES BlogPost(PostID) ON DELETE CASCADE,
    UserId BIGINT NOT NULL REFERENCES BlogUser(UserId) ON DELETE CASCADE,
    Rating SMALLINT NOT NULL CHECK (Rating >= 1 AND Rating <= 5),
    CreatedOn TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    UpdatedOn TIMESTAMP,
    CONSTRAINT UQ_PostRating_User_Post UNIQUE (PostId, UserId)
);

CREATE INDEX IX_PostRating_PostId ON PostRating(PostId);
CREATE INDEX IX_PostRating_UserId ON PostRating(UserId);
```

### Service Pattern

```csharp
public class RatingSvc
{
    private readonly IRatingRepo _ratingRepo;
    
    public Result RatePost(long postId, long userId, int rating)
    {
        if (rating < 1 || rating > 5)
            return Result.Failure("Rating must be between 1 and 5");
            
        var existing = _ratingRepo.GetByPostAndUser(postId, userId);
        if (existing != null)
        {
            existing.Rating = rating;
            existing.UpdatedOn = DateTime.UtcNow;
            _ratingRepo.Update(existing);
        }
        else
        {
            _ratingRepo.Insert(new PostRating
            {
                PostId = postId,
                UserId = userId,
                Rating = rating,
                CreatedOn = DateTime.UtcNow
            });
        }
        return Result.Success();
    }
    
    public (double Average, int Count) GetPostRatingStats(long postId)
    {
        var avg = _ratingRepo.GetAverageByPost(postId);
        var count = _ratingRepo.GetCountByPost(postId);
        return (avg, count);
    }
}
```
