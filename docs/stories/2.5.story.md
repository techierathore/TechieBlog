## Story

**As a** registered user,
**I want** to manage my profile and change my password,
**so that** I can keep my account information current.

3. User can update social links (Twitter, LinkedIn, GitHub)
4. User can upload/change profile image
5. User can change password (requires current password)
6. Form validation with inline error messages
7. Success confirmation after updates

### Code-Behind Pattern

```csharp
public partial class ProfilePage
{
    [Inject] private AuthSvc AuthService { get; set; } = default!;
    [Inject] private AuthenticationStateProvider AuthStateProvider { get; set; } = default!;

    private ProfileModel profileModel = new();
    private PasswordChangeModel passwordModel = new();
    private string profileMessage = string.Empty;
    private bool profileSuccess = false;
    private string passwordMessage = string.Empty;
    private bool passwordSuccess = false;
    private long currentUserId;

    protected override async Task OnInitializedAsync()
    {
        var authState = await AuthStateProvider.GetAuthenticationStateAsync();
        var userIdClaim = authState.User.FindFirst(ClaimTypes.PrimarySid);

        if (userIdClaim != null && long.TryParse(userIdClaim.Value, out var userId))
        {
            currentUserId = userId;
            var user = AuthService.GetUserProfile(userId);
            if (user != null)
            {
                profileModel = new ProfileModel
                {
                    FirstName = user.FirstName,
                    LastName = user.LastName,
                    ProfileDescription = user.ProfileDescription,
                    TwitterUrl = user.TwiiterUrl,  // Note typo in model
                    LinkedInUrl = user.LinkedInUrl,
                    GitHubUrl = user.GitHubUrl
                };
            }
        }
    }

    private async Task SaveProfile()
    {
        profileMessage = string.Empty;

        var result = AuthService.UpdateProfile(currentUserId, profileModel);
        if (result.IsSuccess)
        {
            profileSuccess = true;
            profileMessage = "Profile updated successfully!";
        }
        else
        {
            profileSuccess = false;
            profileMessage = result.ErrorMessage;
        }
    }

    private async Task ChangePassword()
    {
        passwordMessage = string.Empty;

        if (passwordModel.NewPassword != passwordModel.ConfirmPassword)
        {
            passwordSuccess = false;
            passwordMessage = "New passwords do not match";
            return;
        }

        var validationResult = PasswordValidator.Validate(passwordModel.NewPassword);
        if (!validationResult.IsValid)
        {
            passwordSuccess = false;
            passwordMessage = string.Join(". ", validationResult.Errors);
            return;
        }

        var result = AuthService.ChangePassword(
            currentUserId,
            passwordModel.CurrentPassword,
            passwordModel.NewPassword
        );

        if (result.IsSuccess)
        {
            passwordSuccess = true;
            passwordMessage = "Password changed successfully!";
            passwordModel = new PasswordChangeModel(); // Clear form
        }
        else
        {
            passwordSuccess = false;
            passwordMessage = result.ErrorMessage;
        }
    }

    private class ProfileModel
    {
        [Required(ErrorMessage = "First name is required")]
        [StringLength(50, ErrorMessage = "First name cannot exceed 50 characters")]
        public string FirstName { get; set; } = string.Empty;

        [Required(ErrorMessage = "Last name is required")]
        [StringLength(50, ErrorMessage = "Last name cannot exceed 50 characters")]
        public string LastName { get; set; } = string.Empty;

        [StringLength(500, ErrorMessage = "Bio cannot exceed 500 characters")]
        public string ProfileDescription { get; set; } = string.Empty;

        [Url(ErrorMessage = "Invalid Twitter URL")]
        public string TwitterUrl { get; set; } = string.Empty;

        [Url(ErrorMessage = "Invalid LinkedIn URL")]
        public string LinkedInUrl { get; set; } = string.Empty;

        [Url(ErrorMessage = "Invalid GitHub URL")]
        public string GitHubUrl { get; set; } = string.Empty;
    }

    private class PasswordChangeModel
    {
        public string CurrentPassword { get; set; } = string.Empty;
        public string NewPassword { get; set; } = string.Empty;
        public string ConfirmPassword { get; set; } = string.Empty;
    }
}
```

### AuthSvc Extension Pattern

```csharp
public class AuthSvc
{
    public AppUser? GetUserProfile(long userId)
    {
        return UserRepo.GetSingle(userId);
    }

    public Result<bool> UpdateProfile(long userId, ProfileData profileData)
    {
        var user = UserRepo.GetSingle(userId);
        if (user == null)
            return Result<bool>.Failure("User not found");

        user.FirstName = profileData.FirstName;
        user.LastName = profileData.LastName;
        user.ProfileDescription = profileData.ProfileDescription;
        user.TwiiterUrl = profileData.TwitterUrl;
        user.LinkedInUrl = profileData.LinkedInUrl;
        user.GitHubUrl = profileData.GitHubUrl;
        user.UpdatedOn = DateTime.UtcNow;

        UserRepo.Update(user);
        return Result<bool>.Success(true);
    }

    public Result<bool> ChangePassword(long userId, string currentPassword, string newPassword)
    {
        var user = UserRepo.GetSingle(userId);
        if (user == null)
            return Result<bool>.Failure("User not found");

        // Verify current password
        var hashedCurrentPassword = AppEncrypt.CreateHash(currentPassword);
        if (user.LoginPass != hashedCurrentPassword)
            return Result<bool>.Failure("Current password is incorrect");

        // Validate new password
        var validationResult = PasswordValidator.Validate(newPassword);
        if (!validationResult.IsValid)
            return Result<bool>.Failure(string.Join(". ", validationResult.Errors));

        // Update password
        user.LoginPass = AppEncrypt.CreateHash(newPassword);
        user.UpdatedOn = DateTime.UtcNow;
        UserRepo.Update(user);

        return Result<bool>.Success(true);
    }
}
```
