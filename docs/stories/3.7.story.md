# Story 3.7: Series/Collections Feature

## Status

## Story

**As an** author,
**I want** to group related posts into series,
**so that** readers can follow multi-part content in order.

3. Posts can be assigned to a series with part number
4. Series page displays all posts in order
5. Series navigation (prev/next) shown on individual posts
6. All series listing page available
7. Author can reorder posts within a series

### Database Schema

```sql
-- Series table
CREATE TABLE BlogSeries (
    SeriesId BIGSERIAL PRIMARY KEY,
    Name VARCHAR(200) NOT NULL,
    Slug VARCHAR(200) NOT NULL UNIQUE,
    Description TEXT,
    Status VARCHAR(50) NOT NULL DEFAULT 'In Progress', -- 'In Progress', 'Complete'
    AuthorId BIGINT NOT NULL REFERENCES AppUser(UserId),
    CreatedOn TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    UpdatedOn TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP
);

-- Add series fields to Post table
ALTER TABLE Post ADD COLUMN SeriesId BIGINT REFERENCES BlogSeries(SeriesId);
ALTER TABLE Post ADD COLUMN SeriesPartNumber INT;

-- Index for series posts
CREATE INDEX idx_post_series ON Post(SeriesId, SeriesPartNumber);
```

### SeriesSvc Pattern

```csharp
namespace BlogEngine.Services;

public class SeriesSvc
{
    private readonly IBlogSeriesRepo SeriesRepo;
    private readonly IBlogPostRepo PostRepo;

    public SeriesSvc(IBlogSeriesRepo seriesRepo, IBlogPostRepo postRepo)
    {
        SeriesRepo = seriesRepo;
        PostRepo = postRepo;
    }

    public List<BlogSeries> GetAllSeries()
    {
        var series = SeriesRepo.GetAll();
        foreach (var s in series)
        {
            s.PostCount = PostRepo.GetPostCountBySeries(s.SeriesId);
        }
        return series;
    }

    public BlogSeries? GetSeriesBySlug(string slug)
    {
        var series = SeriesRepo.GetBySlug(slug);
        if (series != null)
        {
            series.Posts = PostRepo.GetPostsBySeries(series.SeriesId);
        }
        return series;
    }

    public long CreateSeries(BlogSeries series)
    {
        series.Slug = SlugGenerator.Generate(series.Name);
        series.CreatedOn = DateTime.UtcNow;
        series.UpdatedOn = DateTime.UtcNow;
        return SeriesRepo.Insert(series);
    }

    public void UpdateSeries(BlogSeries series)
    {
        series.UpdatedOn = DateTime.UtcNow;
        SeriesRepo.Update(series);
    }

    public void DeleteSeries(long seriesId)
    {
        // First remove series association from posts
        PostRepo.ClearSeriesFromPosts(seriesId);
        SeriesRepo.Delete(seriesId);
    }

    public int GetNextPartNumber(long seriesId)
    {
        var maxPart = PostRepo.GetMaxPartNumberInSeries(seriesId);
        return maxPart + 1;
    }

    public SeriesNavigation? GetSeriesNavigation(long postId)
    {
        var post = PostRepo.GetSingle(postId);
        if (post?.SeriesId == null) return null;

        var seriesPosts = PostRepo.GetPostsBySeries(post.SeriesId.Value);
        var series = SeriesRepo.GetSingle(post.SeriesId.Value);

        var currentIndex = seriesPosts.FindIndex(p => p.PostID == postId);
        if (currentIndex < 0) return null;

        return new SeriesNavigation
        {
            SeriesName = series?.Name ?? "",
            SeriesSlug = series?.Slug ?? "",
            CurrentPart = post.SeriesPartNumber ?? 0,
            TotalParts = seriesPosts.Count,
            PreviousPost = currentIndex > 0 ? seriesPosts[currentIndex - 1] : null,
            NextPost = currentIndex < seriesPosts.Count - 1 ? seriesPosts[currentIndex + 1] : null
        };
    }
}

public class SeriesNavigation
{
    public string SeriesName { get; set; } = string.Empty;
    public string SeriesSlug { get; set; } = string.Empty;
    public int CurrentPart { get; set; }
    public int TotalParts { get; set; }
    public BlogPost? PreviousPost { get; set; }
    public BlogPost? NextPost { get; set; }
}
```

### ManagePost Series Selector Pattern

```razor
<FluentCard>
    <h3>Series</h3>
    <p class="form-hint">Optionally add this post to a multi-part series.</p>

    <div class="form-row">
        <div class="form-group" style="flex: 2;">
            <FluentSelect Label="Series"
                          TOption="BlogSeries"
                          Items="@availableSeries"
                          OptionText="@(s => s.Name)"
                          OptionValue="@(s => s.SeriesId.ToString())"
                          @bind-SelectedOption="selectedSeries">
                <OptionTemplate>
                    @context.Name (@context.PostCount parts)
                </OptionTemplate>
            </FluentSelect>
        </div>
        <div class="form-group" style="flex: 1;">
            <FluentNumberField Label="Part Number"
                               @bind-Value="PageObj.SeriesPartNumber"
                               Min="1"
                               Disabled="@(selectedSeries == null)" />
        </div>
    </div>

    @if (selectedSeries != null)
    {
        <p>This will be Part @(PageObj.SeriesPartNumber ?? suggestedPartNumber) of "@selectedSeries.Name"</p>
    }
</FluentCard>

@code {
    private List<BlogSeries> availableSeries = new();
    private BlogSeries? selectedSeries;
    private int suggestedPartNumber = 1;

    protected override async Task OnInitializedAsync()
    {
        availableSeries = SeriesSvc.GetAllSeries();

        if (PageObj.SeriesId.HasValue)
        {
            selectedSeries = availableSeries.FirstOrDefault(s => s.SeriesId == PageObj.SeriesId);
        }
    }

    private void OnSeriesChanged(BlogSeries? series)
    {
        selectedSeries = series;
        if (series != null)
        {
            PageObj.SeriesId = series.SeriesId;
            if (!PageObj.SeriesPartNumber.HasValue)
            {
                suggestedPartNumber = SeriesSvc.GetNextPartNumber(series.SeriesId);
                PageObj.SeriesPartNumber = suggestedPartNumber;
            }
        }
        else
        {
            PageObj.SeriesId = null;
            PageObj.SeriesPartNumber = null;
        }
    }
}
```
